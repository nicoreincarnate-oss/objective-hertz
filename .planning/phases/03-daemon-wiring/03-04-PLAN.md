```python
"""Phase 3 Wave 4: Hermes voice session orchestration.

Wires shared.voice.parakeet_client + kokoro_client + intent_router into a
voice-driven interaction loop for the Jarvis war room. Dead code path until
operator starts Parakeet (ASR) and Kokoro (TTS) services on the Studio —
Phase 42.5 v2 operator checklist item.

Flow:
1. Audio input → Parakeet (ASR) → text
2. Text → intent_router → daemon dispatch (e.g. ask titan, ask clawdbot)
3. Daemon response → Kokoro (TTS) → audio output

Opt-in via ENABLE_VOICE_LOOP=true.
"""

from __future__ import annotations

import logging
import os
from typing import Any

logger = logging.getLogger("perseus.hermes.voice")


def voice_loop_enabled() -> bool:
    """Runtime check — only route through voice when operator opts in
    AND both Parakeet and Kokoro services are reachable."""
    if os.environ.get("ENABLE_VOICE_LOOP", "false").lower() != "true":
        return False
    # Additional liveness check happens in handle_voice_input
    return True


async def handle_voice_input(
    audio_bytes: bytes,
    *,
    llm_client: Any,
) -> dict:
    """Process an audio input through the full voice loop.

    Returns a dict with:
      {
        "status": "complete|failed|skipped",
        "transcription": str,
        "response_text": str,
        "response_audio": bytes,
        "daemon_dispatched": str,
        "error": str | None,
      }
    """
    if not voice_loop_enabled():
        return {"status": "skipped", "reason": "ENABLE_VOICE_LOOP=false"}

    try:
        # Lazy import so a missing Parakeet/Kokoro service doesn't crash module load
        from shared.voice.parakeet_client import ParakeetClient
        from shared.voice.kokoro_client import KokoroClient
        from shared.voice.intent_router import route_intent

        asr = ParakeetClient()
        tts = KokoroClient()

        # Liveness check — if services aren't running, fall through cleanly
        if not await asr.is_available() or not await tts.is_available():
            return {
                "status": "skipped",
                "reason": "Parakeet or Kokoro service not reachable",
            }

        transcription = await asr.transcribe(audio_bytes)
        if not transcription:
            return {"status": "failed", "error": "Empty transcription from Parakeet"}

        intent = await route_intent(
            transcription,
            llm_client=llm_client,
        )
        daemon = intent.get("daemon", "hermes")
        response_text = intent.get("response_text", "")

        if not response_text:
            return {
                "status": "failed",
                "transcription": transcription,
                "error": "Intent router returned empty response",
            }

        response_audio = await tts.synthesize(response_text)

        return {
            "status": "complete",
            "transcription": transcription,
            "response_text": response_text,
            "response_audio": response_audio,
            "daemon_dispatched": daemon,
            "error": None,
        }

    except ImportError as exc:
        # Parakeet/Kokoro clients not installed — dead path, expected until operator installs
        logger.debug("Voice loop modules not fully wired: %s", exc)
        return {
            "status": "skipped",
            "reason": f"Voice modules not available: {exc}",
        }
    except Exception as exc:
        logger.exception("Voice loop exception: %s", exc)
        return {"status": "failed", "error": str(exc)}
```
