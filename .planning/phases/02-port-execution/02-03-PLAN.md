```python
"""P0-3: Subprocess wrapper for macOS sandbox-exec verifier runs.

Wraps test execution in the sandbox profile at
litellm/sandboxes/verifier.sb. Ensures untrusted LLM-generated patches
can't escape to touch ~/.ssh, wallets, or env files.

Threat model addressed: prompt-injected Ruflo fix that tries to run
`rm -rf ~/.ssh` or exfiltrate Conway wallets. Sandbox denies network
egress and home-dir reads outside the scratch worktree.
"""

import os
import subprocess
from pathlib import Path

SANDBOX_PROFILE = Path(__file__).parent.parent.parent / "litellm" / "sandboxes" / "verifier.sb"


class SandboxRunner:
    """Run tests inside macOS sandbox-exec with the verifier profile."""

    def __init__(self, profile_path: Path | None = None):
        self.profile = profile_path or SANDBOX_PROFILE
        if not self.profile.exists():
            raise FileNotFoundError(f"Sandbox profile not found: {self.profile}")

    def run(self, command: list[str], cwd: Path | None = None, timeout_s: int = 60) -> subprocess.CompletedProcess:
        """Execute a command inside the sandbox.

        CRITICAL: passes -D HOME=$HOME so (param "HOME") in the profile
        resolves correctly. Without this the deny rules silently no-op.
        """
        full_cmd = [
            "/usr/bin/sandbox-exec",
            "-D", f"HOME={os.environ['HOME']}",
            "-f", str(self.profile),
        ] + command
        return subprocess.run(
            full_cmd,
            cwd=cwd,
            capture_output=True,
            text=True,
            timeout=timeout_s,
        )

    def run_pytest(self, test_path: Path, cwd: Path | None = None, timeout_s: int = 60) -> subprocess.CompletedProcess:
        """Convenience: run pytest on a specific path inside the sandbox."""
        return self.run(
            ["python3", "-m", "pytest", str(test_path), "-q", "--tb=short"],
            cwd=cwd,
            timeout_s=timeout_s,
        )
```
